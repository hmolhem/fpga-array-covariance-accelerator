% --- Page Geometry Configuration ---
\geometry{
    a4paper,
    top=2.5cm,
    bottom=2.5cm,
    left=3.0cm,
    right=2.5cm,
    headheight=16pt,
    includeheadfoot
}

% --------------------------------------------------
% Header fallback definitions
% --------------------------------------------------
% These commands are normally defined in main.tex.
% The fallbacks make this setup file portable.
\providecommand{\courseshorttitle}{Engineering Project}
\providecommand{\reportshorttitle}{Technical Report}
\providecommand{\headerleft}{\courseshorttitle}
\providecommand{\headerright}{\reportshorttitle}

% --------------------------------------------------
% Headers and Footers
% --------------------------------------------------
% The report class applies the "plain" style on chapter-opening pages,
% the first page of the table of contents, lists, and bibliography.
% Defining both styles guarantees a consistent header throughout.
\pagestyle{fancy}
\fancyhf{}

\fancyhead[L]{\small\slshape\nouppercase{\headerleft}}
\fancyhead[R]{\small\slshape\nouppercase{\headerright}}
\fancyfoot[C]{\thepage}

\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\setlength{\headheight}{16pt}

\fancypagestyle{plain}{%
    \fancyhf{}%
    \fancyhead[L]{\small\slshape\nouppercase{\headerleft}}%
    \fancyhead[R]{\small\slshape\nouppercase{\headerright}}%
    \fancyfoot[C]{\thepage}%
    \renewcommand{\headrulewidth}{0.4pt}%
    \renewcommand{\footrulewidth}{0pt}%
}

% Cover-page style: no header or footer.
\fancypagestyle{titlepage}{%
    \fancyhf{}%
    \renewcommand{\headrulewidth}{0pt}%
    \renewcommand{\footrulewidth}{0pt}%
}

% --- Captions, Paragraphs, and Spacing ---
\captionsetup{margin=10pt, font=small, labelfont=bf}
\setlength{\parindent}{0em}
\setlength{\parskip}{0.65em}
\allowdisplaybreaks

% --- List and Numbering Customization ---
\renewcommand{\listfigurename}{List of Figures}
\renewcommand{\listtablename}{List of Tables}
\setcounter{tocdepth}{3}
\setcounter{secnumdepth}{3}
